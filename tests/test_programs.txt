// Program 3: Control Flow (If / Else / While / For)
class ControlFlow {
    void evaluate(int limit) {
        int i = 0;
        while (i < limit) {
            if (i == 5) {
                return;
            } else {
                i = i + 1;
            }
        }
        for (i = 0; i <= 10; i = i + 1) {
            limit = limit * 2;
        }
    }
}
